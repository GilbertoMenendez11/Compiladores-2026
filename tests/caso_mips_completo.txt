def factorial(n):
    if n <= 1:
        return 1
    else:
        return n * factorial(n - 1)

def suma_pares(maximo):
    suma = 0
    i = 0
    while i <= maximo:
        if i % 2 == 0:
            suma = suma + i
        i = i + 1
    return suma
