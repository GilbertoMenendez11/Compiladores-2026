def factorial(n):
    if n <= 1:
        return 1
    else:
        return n * factorial(n - 1)

def sumaPares(max):
    suma = 0
    i = 0
    while i <= max:
        if (i - (i / 2) * 2) == 0:
            suma = suma + i
        i = i + 1
    return suma
