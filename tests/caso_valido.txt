/* Programa de prueba válido
   Calcula factorial y suma de pares */

int limite = 10;
float tasa = 0.5, total;
string mensaje = "Resultado: ";

int factorial(int n) {
    if (n <= 1) {
        return 1;
    } else {
        return n * factorial(n - 1);
    }
}

int sumaPares(int max) {
    int suma = 0;
    int i;
    for (i = 0; i <= max; i = i + 1) {
        if (i - (i / 2) * 2 == 0)
            suma = suma + i;
    }
    return suma;
}
